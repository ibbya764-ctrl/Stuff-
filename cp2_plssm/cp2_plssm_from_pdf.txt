# ─────────────────────────────────────────────────────────────────────────────
# § GELL-MANN MATRICES  (SU(3) generators; fixed, not learned)
# ─────────────────────────────────────────────────────────────────────────────
def _build_gell_mann() -> torch.Tensor:
    """Returns the 8 Gell-Mann matrices as a (8,3,3) complex64 tensor."""
    lam = torch.zeros(8, 3, 3, dtype=torch.complex64)
    # λ₁
    lam[0, 0, 1] = 1;  lam[0, 1, 0] = 1
    # λ₂
    lam[1, 0, 1] = -1j; lam[1, 1, 0] = 1j
    # λ₃
    lam[2, 0, 0] = 1;  lam[2, 1, 1] = -1
    # λ₄
    lam[3, 0, 2] = 1;  lam[3, 2, 0] = 1
    # λ₅
    lam[4, 0, 2] = -1j; lam[4, 2, 0] = 1j
    # λ₆
    lam[5, 1, 2] = 1;  lam[5, 2, 1] = 1
    # λ₇
    lam[6, 1, 2] = -1j; lam[6, 2, 1] = 1j
    # λ₈  (diagonal)
    s3 = 1.0 / math.sqrt(3.0)
    lam[7, 0, 0] = s3;  lam[7, 1, 1] = s3;  lam[7, 2, 2] = -2 * s3
    return lam
GELL_MANN = _build_gell_mann()   # (8, 3, 3) complex — registered as buffer below
# ─────────────────────────────────────────────────────────────────────────────
# § CP² UTILITY FUNCTIONS
# ─────────────────────────────────────────────────────────────────────────────
def cp2_normalise(z: torch.Tensor) -> torch.Tensor:
    """Normalise homogeneous coordinates: z / ‖z‖.  z: (..., 3) complex."""
    return z / z.norm(dim=-1, keepdim=True).clamp(min=1e-8)
def fubini_study_distance(z1: torch.Tensor, z2: torch.Tensor) -> torch.Tensor:
    """
    Fubini–Study geodesic distance d(z1,z2) = arccos |⟨z1,z2⟩|.
    z1, z2: (..., 3) complex, unit-norm.  Returns (...,) real ∈ [0, π/2].
    """
    inner = (z1.conj() * z2).sum(dim=-1).abs().clamp(max=1.0 - 1e-7)
    return torch.acos(inner)
def cremona_involution(z: torch.Tensor) -> torch.Tensor:
    """
    Real Cremona involution f[z₀:z₁:z₂] = [z̄₁z̄₂ : z̄₀z̄₂ : z̄₀z̄₁].
    Maps matter branch (|z|=φ) ↔ antimatter branch (|z|=1/φ).
    z: (..., 3) complex.
    """
    z0, z1, z2 = z[..., 0], z[..., 1], z[..., 2]
    w = torch.stack([z1.conj() * z2.conj(),
                     z0.conj() * z2.conj(),
                     z0.conj() * z1.conj()], dim=-1)
    return cp2_normalise(w)
def branch_magnitude(z: torch.Tensor) -> torch.Tensor:
    """
    Kerr–Newman branch parameter |z_eff| = |z₁| / |z₀|.
    > φ: super-extremal / matter;  < 1/φ: sub-extremal / antimatter.
    """
    return z[..., 1].abs() / z[..., 0].abs().clamp(min=1e-8)
def phi_vacuum(device: torch.device) -> torch.Tensor:
    """φ-vacuum p_φ = [1:φ:0] (normalised) — unique SL(2,Z)-stable fixed point."""
    p = torch.tensor([1.0, PHI, 0.0], dtype=torch.complex64, device=device)
    return cp2_normalise(p)
def toric_fixed_points(device: torch.device) -> torch.Tensor:
    """Three toric fixed points P₀,P₁,P₂ = identity rows. Shape (3,3) complex."""
    return torch.eye(3, dtype=torch.complex64, device=device)
# ─────────────────────────────────────────────────────────────────────────────
# § SU(3) EQUIVARIANT TRANSFORM
#   Replaces unconstrained linear updates with proper group-action steps.
#   State evolves as z' = exp(i Σ cₖ λₖ) · z ∈ SU(3)/U(2) = CP².
# ─────────────────────────────────────────────────────────────────────────────
class SU3Transform(nn.Module):
    """
    Evolves a CP² point via a context-dependent SU(3) element.
    M = exp(i · Σ_k  c_k(x) · λ_k)   where λ_k are Gell-Mann matrices.
    This is the *only* correct way to update a state on CP² while
    preserving the coset structure SU(3)/U(2) (§3.1, §6.1 of paper).
    All eight generators are used; the network predicts the coefficients.
    """
    def __init__(self, context_dim: int):
        super().__init__()
        self.coeff_net = nn.Sequential(
            nn.Linear(context_dim, 64),
            nn.SiLU(),
            nn.Linear(64, 8),   # 8 Gell-Mann coefficients
        )
        # Learnable temperature: small init → near-identity at start
        self.log_temp = nn.Parameter(torch.tensor(-2.0))
    def forward(self, context: torch.Tensor, z: torch.Tensor) -> torch.Tensor:
        """
        context : (..., context_dim)  real
        z        : (..., 3)           complex, unit-norm
        Returns  : (..., 3)           complex, unit-norm
        """
        device = context.device
        gm = GELL_MANN.to(device)                          # (8,3,3) complex
        coeffs = self.coeff_net(context) * self.log_temp.exp()  # (..., 8)
        # Lie-algebra element:  A = i Σ c_k λ_k  (anti-Hermitian ⟹ exp(A) ∈ U(3))
        A = torch.einsum('...k,kij->...ij',
                          coeffs.to(torch.complex64), 1j * gm)   # (...,3,3)
        M = torch.matrix_exp(A)                            # (...,3,3) ∈ U(3)≈SU(3)
        z_new = torch.einsum('...ij,...j->...i', M, z)
        return cp2_normalise(z_new)
# ─────────────────────────────────────────────────────────────────────────────
# § 2+2 CAUSAL SPLIT   TM = E ⊕ E⊥
#   Implements Theorem 2.1 / Axioms 7–8.
#   E: rank-2 Kähler-active subbundle where J² = −1.
#   E⊥: causal complement where J = 0.
# ─────────────────────────────────────────────────────────────────────────────
class CausalSplit22(nn.Module):
    """
    Canonical 2+2 decomposition of the hidden state.
    The Ω-compatibility condition  Ω_µρ Ω^ρ_ν = −ℓ²Π_µν  forces an exact
    splitting.  Here we implement it as two parallel streams that interact
    only through a learned scale ℓ², enforcing the same algebraic constraint.
    E-stream:   carries phase / geometric information (J acts here)
    E⊥-stream:  carries causal / sequential information (J = 0)
    """
    def __init__(self, dim: int):
        super().__init__()
        assert dim % 4 == 0, "dim must be divisible by 4 for 2+2 split"
        self.half = dim // 2
        self.qtr  = dim // 4
        self.proj_E     = nn.Linear(dim, self.half)
        self.proj_Eperp = nn.Linear(dim, self.half)
        self.merge      = nn.Linear(dim, dim)
        # Learnable ℓ²; initialised to 1 (unit scale)
        self.log_ell_sq = nn.Parameter(torch.zeros(1))
    def _apply_J(self, v: torch.Tensor) -> torch.Tensor:
        """
        Apply J on E: rotate 90° between consecutive dimension pairs.
        (v_re, v_im) → (−v_im, v_re).   J² = −I by construction.
        """
        re, im = v[..., :self.qtr], v[..., self.qtr:]
        return torch.cat([-im, re], dim=-1)
    def forward(self, h: torch.Tensor):
        """
        h: (..., dim)
        Returns (h_E, h_Eperp, h_merged, Jh_E)
        """
        h_E     = self.proj_E(h)
        h_Eperp = self.proj_Eperp(h)
        Jh_E    = self._apply_J(h_E) * self.log_ell_sq.exp()
        h_merged = self.merge(torch.cat([h_E, h_Eperp], dim=-1))
        return h_E, h_Eperp, h_merged, Jh_E
# ─────────────────────────────────────────────────────────────────────────────
# § DUAL-BRANCH EMBEDDING   (matter φ-branch + antimatter 1/φ-branch)
#   §5.1 of paper: matter at |z|=φ, antiparticles at |z|=1/φ, photons at |z|=1.
#   The Cremona involution f maps between branches.
# ─────────────────────────────────────────────────────────────────────────────
class DualBranchEmbedding(nn.Module):
    """
    Each token simultaneously has a presence in both CP² branches.
    A learned mixing weight α ∈ [0,1] per head determines how much of
    each branch contributes; φ-branch dominates by default (stable vacuum).
    """
    def __init__(self, vocab_size: int, n_heads: int):
        super().__init__()
        self.n_heads = n_heads
        self.embed_re   = nn.Embedding(vocab_size, n_heads * 3)
        self.embed_im   = nn.Embedding(vocab_size, n_heads * 3)
        self.branch_mix = nn.Embedding(vocab_size, n_heads)   # logits → α
    def _push_to_branch(self, z: torch.Tensor, target: float) -> torch.Tensor:
        """Soft rescaling of |z₁/z₀| toward `target` (φ or 1/φ)."""
        z0 = z[..., 0:1]; z1 = z[..., 1:2]; z2 = z[..., 2:3]
        curr = z1.abs() / z0.abs().clamp(min=1e-8)
        scale = (target / curr.clamp(min=1e-8)).detach().clamp(0.3, 3.0)
        return cp2_normalise(torch.cat([z0, z1 * scale, z2], dim=-1))
    def forward(self, tokens: torch.Tensor):
        """
        tokens : (B, T)
        Returns z_phi, z_anti, alpha — shapes (B,T,H,3), (B,T,H,3), (B,T,H)
        """
        B, T = tokens.shape
        H = self.n_heads
        re  = self.embed_re(tokens).view(B, T, H, 3)
        im  = self.embed_im(tokens).view(B, T, H, 3)
        z   = cp2_normalise(torch.complex(re, im))
        z_phi  = self._push_to_branch(z, PHI)
        z_anti = cremona_involution(z_phi.view(-1, 3)).view(B, T, H, 3)
        alpha  = torch.sigmoid(self.branch_mix(tokens))        # (B, T, H)
        return z_phi, z_anti, alpha
# ─────────────────────────────────────────────────────────────────────────────
# § INSTANTON GATE  (new — not in Gemini suggestions)
#   Fermion masses ~ exp(−nπ²) · Z_ψ (§11).
#   Here: n information channels, each suppressed at its instanton rate.
#   This builds a geometric scale hierarchy into the gating mechanism.
# ─────────────────────────────────────────────────────────────────────────────
class InstantonGate(nn.Module):
    """
    Multi-scale gate with fixed geometric suppression exp(−n·π²).
    Channel n carries information "at instanton scale n".
    The network learns *which* channel to activate; the relative weights
    are permanently fixed by the CP² instanton geometry.
    n=1 (heaviest suppression):  ~ 5×10⁻⁵   (top-quark scale)
    n=4 (lighter):               ~ e^{−4π²}  (τ,c,b,t sector)
    n=5 (lightest in hierarchy): ~ e^{−5π²}  (e,u,d sector)
    """
    def __init__(self, dim: int, n_levels: int = 5):
        super().__init__()
        weights = torch.tensor(
            [math.exp(-n * S_YM) for n in range(1, n_levels + 1)],
            dtype=torch.float32
        )
        # Normalise so they sum to 1 (softmax-like baseline)
        self.register_buffer('inst_weights', weights / weights.sum())
        self.gate_net    = nn.Linear(dim, n_levels)
        self.chan_projs  = nn.ModuleList([nn.Linear(dim, dim) for _ in range(n_levels)])
    def forward(self, h: torch.Tensor) -> torch.Tensor:
        gates = F.softmax(self.gate_net(h), dim=-1)                # (..., N)
        gates = gates * self.inst_weights                          # geometric weighting
        gates = gates / gates.sum(dim=-1, keepdim=True).clamp(1e-8)
        out   = sum(gates[..., n:n+1] * proj(h)
                    for n, proj in enumerate(self.chan_projs))
        return out
# ─────────────────────────────────────────────────────────────────────────────
# § MACHIN GATE  (new — not in Gemini suggestions)
#   arctan(½) + arctan(⅓) = π/4   [D20 / §9]
#   arctan(½) = d_FS(φ, 1/φ):  inter-branch distance
#   arctan(⅓):  Machin complement; origin of charm-quark δ
#
#   Two parallel branches with fixed phase angles that compose exactly
#   to π/4 — a natural quarter-period gate with geometric grounding.
# ─────────────────────────────────────────────────────────────────────────────
class MachinGate(nn.Module):
    """
    Two-branch gate whose phases sum to exactly π/4.
    Branch A (arctan½): inter-branch information pathway
    Branch B (arctan⅓): Machin-complement / charm-scale pathway
    Output = proj( h · sin(g_A + g_B) )
    Since g_A→arctan½ and g_B→arctan⅓ when saturated,
    the composed gate saturates at sin(π/4) = 1/√2 — a natural ceiling
    that prevents runaway activations without any explicit clipping.
    """
    def __init__(self, dim: int):
        super().__init__()
        self.branch_A = nn.Linear(dim, dim)
        self.branch_B = nn.Linear(dim, dim)
        self.out_proj = nn.Linear(dim, dim)
        self.register_buffer('phase_A', torch.tensor(ARCTAN_HALF,  dtype=torch.float32))
        self.register_buffer('phase_B', torch.tensor(ARCTAN_THIRD, dtype=torch.float32))
    def forward(self, h: torch.Tensor) -> torch.Tensor:
        gA = torch.tanh(self.branch_A(h)) * self.phase_A   # bounded by arctan½
        gB = torch.tanh(self.branch_B(h)) * self.phase_B   # bounded by arctan⅓
        composed = gA + gB                                  # → max ≈ π/4
        return self.out_proj(h * torch.sin(composed))       # sin(π/4) = 1/√2 ceiling
# ─────────────────────────────────────────────────────────────────────────────
# § GOLDEN ATTRACTOR  (new — not in Gemini suggestions)
#   §6.3: φ is the unique stable positive fixed point of z ↦ 1 + 1/z.
#   We use this as a self-normalising map on hidden state norms:
#   iterating z ↦ 1 + 1/z contracts toward φ regardless of initialisation.
#   No explicit gradient clipping needed — the geometry does the work.
# ─────────────────────────────────────────────────────────────────────────────
class GoldenAttractor(nn.Module):
    """
    Soft attractor that draws hidden-state norms toward φ via the
    continued-fraction map z ↦ 1 + 1/z.
    Three iterations are sufficient to contract within 1% of φ for
    any initial z ∈ [0.1, 10].  The attraction strength α is learned.
    """
    def __init__(self, n_iter: int = 3):
        super().__init__()
        self.n_iter = n_iter
        self.log_alpha = nn.Parameter(torch.tensor(-2.3))   # α ≈ 0.1 initially
    def forward(self, h: torch.Tensor) -> torch.Tensor:
        norm = h.norm(dim=-1, keepdim=True).clamp(min=1e-8)
        unit = h / norm
        mag  = norm / PHI                                    # normalise to φ scale
        for _ in range(self.n_iter):
            mag = 1.0 + 1.0 / mag.clamp(min=1e-4)          # z ↦ 1 + 1/z
        alpha = torch.sigmoid(self.log_alpha)
        final_norm = (1 - alpha) * norm + alpha * (mag * PHI)
        return unit * final_norm.clamp(max=PHI * 20)
# ─────────────────────────────────────────────────────────────────────────────
# § GUE SPECTRAL REGULARISER  (new — not in Gemini suggestions)
#   Theorem 15.2: σ(φ)/σ(1/φ) = φ²  (spacing ratio between branches).
#   Theorem 15.1: the framework belongs to GUE symmetry class (β=2).
#   We enforce the φ² spacing ratio on singular-value spectra of weights.
# ─────────────────────────────────────────────────────────────────────────────
class GUESpectralRegulariser(nn.Module):
    """
    Auxiliary loss that penalises deviations of weight-matrix singular-value
    spacings from the φ² ratio predicted by Theorem 15.2.
    """
    def __init__(self):
        super().__init__()
        self.register_buffer('log_target', torch.tensor(math.log(PHI_SQ)))
    def forward(self, W: torch.Tensor) -> torch.Tensor:
        if W.dim() != 2 or min(W.shape) < 4:
            return W.new_zeros(())
        try:
            sv = torch.linalg.svdvals(W.float()).sort(descending=True).values
        except Exception:
            return W.new_zeros(())
        diffs = (sv[:-1] - sv[1:]).clamp(min=1e-8)
        mid   = len(diffs) // 2
        sigma_hi = diffs[:mid].mean()
        sigma_lo = diffs[mid:].mean().clamp(min=1e-8)
        log_ratio = (sigma_hi / sigma_lo).log()
        return (log_ratio - self.log_target) ** 2
# ─────────────────────────────────────────────────────────────────────────────
# § TORIC GENERATION MIXER  (new)
#   §10: CP² has exactly 3 toric fixed points P₀,P₁,P₂ corresponding to
#   fermion generations 1,2,3.
#   Each of the 3 CP² heads is attracted to one fixed point;
#   generation weights are the FS-proximity softmax across heads.
# ─────────────────────────────────────────────────────────────────────────────
class ToricGenerationMixer(nn.Module):
    """
    Mixes information across the 3 CP² heads using proximity to each
    toric fixed point as the mixing weight.
    Head 0 ↔ P₀=[1:0:0] ↔ Generation 1 (e, u, d)
    Head 1 ↔ P₁=[0:1:0] ↔ Generation 2 (µ, c, s)
    Head 2 ↔ P₂=[0:0:1] ↔ Generation 3 (τ, t, b)
    """
    def __init__(self, n_heads: int, out_dim: int):
        super().__init__()
        assert n_heads == 3, "ToricGenerationMixer requires exactly 3 heads"
        self.out_proj = nn.Linear(n_heads * 6, out_dim)   # 6 real per head (3 complex)
    def forward(self, z: torch.Tensor) -> torch.Tensor:
        """
        z : (..., 3, 3) complex — (batch, heads, cp2_coords)
        Returns (..., out_dim) real
        """
        device = z.device
        P = toric_fixed_points(device)                      # (3,3) complex
        # FS proximity of each head to its assigned fixed point
        gen_weights = torch.stack([
            fubini_study_distance(z[..., h, :], P[h].expand_as(z[..., h, :]))
            for h in range(3)
        ], dim=-1)                                          # (..., 3) distances
        # Closer ↔ higher weight: use negative distance softmax
        gen_weights = F.softmax(-gen_weights, dim=-1)       # (..., 3)
        # Flatten heads to real
        z_flat = z.view(*z.shape[:-2], -1)                  # (..., 9) complex
        z_real = torch.cat([z_flat.real, z_flat.imag], dim=-1)  # (..., 18)
        # Weight each head's contribution
        out = self.out_proj(z_real)                         # (..., out_dim)
        return out, gen_weights
# ─────────────────────────────────────────────────────────────────────────────
# § BRANCH CROSSING MONITOR  (new)
#   §5.2: critical velocity v_c = c/√φ; at this speed γ = φ and the
#   effective Kerr–Newman parameter crosses |z_eff|=1 (the photon locus).
#   Here we detect when the CP² state crosses the branch boundary and
#   raise an auxiliary loss to discourage spurious crossings.
# ─────────────────────────────────────────────────────────────────────────────
class BranchCrossingMonitor(nn.Module):
    """
    Detects and penalises unintended matter↔antimatter branch transitions.
    A crossing happens when branch_magnitude crosses from >1 to <1 or vice versa.
    The penalty is soft (smooth hinge), not hard-coded.
    """
    def __init__(self, crossing_penalty: float = 0.05):
        super().__init__()
        self.penalty = crossing_penalty
    def forward(self, z_seq: torch.Tensor) -> torch.Tensor:
        """
        z_seq : (B, T, 3) complex — sequence of CP² states
        Returns scalar penalty for spurious branch crossings.
        """
        mag = branch_magnitude(z_seq)                       # (B, T)
        # sign = +1 if matter, -1 if antimatter
        signs = torch.sign(mag - 1.0)
        # Crossing: consecutive tokens with opposite signs
        crossings = (signs[:, 1:] * signs[:, :-1]).clamp(max=0.0).abs()  # 1 at crossings
        return self.penalty * crossings.mean()
# ─────────────────────────────────────────────────────────────────────────────
# § CP² GEODESIC LOSS
#   Uses exact Fubini–Study distances (arccos |⟨z₁,z₂⟩|).
#   Extended with branch-aware weighting:
#     φ-branch predictions weighted by φ²  (stable vacuum → high confidence)
#     1/φ-branch weighted by 1             (unstable → lower confidence)
# ─────────────────────────────────────────────────────────────────────────────
class CP2GeodesicLoss(nn.Module):
    """Branch-aware Fubini–Study geodesic loss."""
    def __init__(self):
        super().__init__()
    def forward(self, z_pred: torch.Tensor, z_tgt: torch.Tensor,
                branch_labels: Optional[torch.Tensor] = None) -> torch.Tensor:
        dist = fubini_study_distance(z_pred, z_tgt)
        if branch_labels is not None:
            w = torch.where(branch_labels > 0,
                            z_pred.new_full(dist.shape, PHI_SQ),
                            torch.ones_like(dist))
            return (w * dist).mean()
        return dist.mean()

# § CP² SSM CELL
# ─────────────────────────────────────────────────────────────────────────────
class CP2SSMCell(nn.Module):
    """
    One recurrent cell of the CP²-Projective–Lorentzian SSM.
    State = (h ∈ ℝ^dim,   z ∈ CP²^{n_heads})
    Update steps
    ────────────
    1. Read CP² coordinates → real info vector
    2. GRU-style gate over (h, x, cp2_info)
    3. 2+2 causal split on gated state
    4. InstantonGate  — geometric scale hierarchy
    5. MachinGate     — Machin composition law
    6. GoldenAttractor — φ continued-fraction normalisation
    7. SU3Transform   — evolve CP² state via SU(3) action
    """
    def __init__(self, x_dim: int, h_dim: int, n_heads: int = 3):
        super().__init__()
        self.h_dim   = h_dim
        self.n_heads = n_heads
        # CP² → real readout
        self.cp2_read = nn.Linear(n_heads * 6, h_dim)
        # Input projection
        self.x_proj = nn.Linear(x_dim, h_dim)
        # GRU gates
        self.gate_u = nn.Linear(h_dim * 2, h_dim)
        self.gate_r = nn.Linear(h_dim * 2, h_dim)
        self.gate_c = nn.Linear(h_dim * 2, h_dim)
        # Geometric modules
        self.causal_split  = CausalSplit22(h_dim)
        self.instanton     = InstantonGate(h_dim)
        self.machin        = MachinGate(h_dim)
        self.phi_attract   = GoldenAttractor()
        self.su3           = SU3Transform(x_dim + h_dim)
        self.ln = nn.LayerNorm(h_dim)
    def _cp2_to_real(self, z: torch.Tensor) -> torch.Tensor:
        """z: (..., H, 3) complex  →  (..., H*6) real"""
        B = z.shape[0]
        zf = z.view(B, -1)                                  # (..., H*3) complex
        return torch.cat([zf.real, zf.imag], dim=-1)        # (..., H*6) real
    def forward(self, x: torch.Tensor, h: torch.Tensor,
                z: torch.Tensor):
        """
        x : (B, x_dim)   — current token embedding
        h : (B, h_dim)   — previous hidden state
        z : (B, H, 3)    — previous CP² coordinates (complex)
        Returns (h_new, z_new)
        """
        B = x.shape[0]
        # ── CP² readout ──────────────────────────────────────────────
        cp2_real = self._cp2_to_real(z)                     # (B, H*6)
        cp2_info = self.cp2_read(cp2_real)                  # (B, h_dim)
        x_p      = self.x_proj(x)                           # (B, h_dim)
        ctx      = x_p + cp2_info
        # ── GRU ──────────────────────────────────────────────────────
        cat_hc = torch.cat([h, ctx], dim=-1)
        u = torch.sigmoid(self.gate_u(cat_hc))
        r = torch.sigmoid(self.gate_r(cat_hc))
        c = torch.tanh(self.gate_c(torch.cat([r * h, ctx], dim=-1)))
        h_gru = (1 - u) * h + u * c
        # ── Geometric processing ──────────────────────────────────────
        _, _, h_split, _ = self.causal_split(h_gru)
        h_inst  = self.instanton(h_split)
        h_mach  = self.machin(h_inst)
        h_phi   = self.phi_attract(h_mach)
        h_new   = self.ln(h_phi)
        # ── SU(3) CP² update ─────────────────────────────────────────
        full_ctx = torch.cat([x, h_new[..., :x.shape[-1]]], dim=-1)   # (B, x_dim+h_dim) — note: h_new may be wider; we slice x_dim for SU3
        # Ensure exactly x_dim + h_dim
        full_ctx = torch.cat([x, h_new], dim=-1)           # (B, x_dim+h_dim)
        z_new = torch.stack([
            self.su3(full_ctx, z[:, head, :])
            for head in range(self.n_heads)
        ], dim=1)                                           # (B, H, 3)
        return h_new, z_new

# § FULL MODEL
# ─────────────────────────────────────────────────────────────────────────────
class CP2ProjectiveSSM(nn.Module):
    """
    CP² Projective–Lorentzian State Space Model (full).
    Hyper-parameters
    ────────────────
    vocab_size   : int   — tokeniser vocabulary
    h_dim        : int   — hidden dimension (must be divisible by 4)
    n_layers     : int   — number of CP²SSMCell layers
    x_dim        : int   — token embedding dimension
    n_heads      : int   — CP² heads (should be 3: three toric fixed points)
    dropout      : float
    Outputs
    ───────
    logits       : (B, T, vocab_size)
    aux_losses   : dict of scalar tensors
    """
    def __init__(self,
                 vocab_size: int,
                 h_dim:      int   = 256,
                 n_layers:   int   = 6,
                 x_dim:      int   = 64,
                 n_heads:    int   = 3,
                 dropout:    float = 0.1):
        super().__init__()
        self.vocab_size = vocab_size
        self.h_dim      = h_dim
        self.n_layers   = n_layers
        self.n_heads    = n_heads
        self.x_dim      = x_dim
        # ── Embedding ────────────────────────────────────────────────
        self.token_embed = nn.Embedding(vocab_size, x_dim)
        self.cp2_embed   = DualBranchEmbedding(vocab_size, n_heads)
        # ── Recurrent layers ─────────────────────────────────────────
        self.cells = nn.ModuleList([
            CP2SSMCell(x_dim, h_dim, n_heads) for _ in range(n_layers)
        ])
        # ── Inter-layer projections ───────────────────────────────────
        self.inter_proj = nn.ModuleList([
            nn.Linear(h_dim, x_dim) for _ in range(n_layers - 1)
        ])
        # ── Toric generation mixer ────────────────────────────────────
        assert n_heads == 3, "n_heads must be 3 (toric fixed points)"
        self.gen_mixer = ToricGenerationMixer(n_heads=3, out_dim=h_dim)
        # ── Output ───────────────────────────────────────────────────
        self.out_proj  = nn.Linear(h_dim + n_heads * 6, h_dim)
        self.out_head  = nn.Linear(h_dim, vocab_size)
        # ── Auxiliary modules ────────────────────────────────────────
        self.geodesic_loss   = CP2GeodesicLoss()
        self.gue_reg         = GUESpectralRegulariser()
        self.branch_monitor  = BranchCrossingMonitor()
        self.drop = nn.Dropout(dropout)
    # ── State initialisation ────────────────────────────────────────
    def init_state(self, B: int, device):
        """Initialise hidden states at the φ-vacuum."""
        h_list = [torch.zeros(B, self.h_dim, device=device)
                  for _ in range(self.n_layers)]
        pv     = phi_vacuum(device)                         # (3,) complex
        z_list = [pv.unsqueeze(0).unsqueeze(0)
                    .expand(B, self.n_heads, 3).clone()
                  for _ in range(self.n_layers)]
        return h_list, z_list
    # ── Single step ─────────────────────────────────────────────────
    def step(self, token: torch.Tensor, h_list, z_list):
        """
        token   : (B,)  integer
        Returns (logits (B,V), h_list_new, z_list_new, aux_dict)
        """
        B = token.shape[0]
        # Embed
        x   = self.token_embed(token)                       # (B, x_dim)
        z_phi, z_anti, alpha = self.cp2_embed(token.unsqueeze(1))
        z_phi  = z_phi.squeeze(1)                           # (B, H, 3)
        z_anti = z_anti.squeeze(1)
        alpha  = alpha.squeeze(1)                           # (B, H)
        # Mix branches (φ-branch dominant)
        z_in = cp2_normalise(
            alpha.unsqueeze(-1) * z_phi +
            (1 - alpha).unsqueeze(-1) * z_anti
        )
        # Process layers
        h_new_list = []; z_new_list = []
        x_cur = x
        for i, cell in enumerate(self.cells):
            h_n, z_n = cell(x_cur, h_list[i], z_list[i])
            h_new_list.append(h_n); z_new_list.append(z_n)
            if i < self.n_layers - 1:
                x_cur = self.inter_proj[i](h_n) + x            # residual skip
        h_fin = h_new_list[-1]
        z_fin = z_new_list[-1]
        # Toric generation mixing
        gen_feat, gen_weights = self.gen_mixer(z_fin)       # (B, h_dim), (B, 3)
        # Readout
        z_real = torch.cat([z_fin.view(B, -1).real,
                             z_fin.view(B, -1).imag], dim=-1)  # (B, H*6)
        out   = self.out_proj(torch.cat([h_fin + gen_feat, z_real], dim=-1))
        logit = self.out_head(self.drop(F.gelu(out)))
        aux = dict(
            z_final      = z_fin,
            gen_weights  = gen_weights,
            alpha        = alpha,
            branch_mag   = branch_magnitude(z_fin[:, 0, :]),
            branch_label = torch.sign(branch_magnitude(z_fin[:, 0, :]) - 1.0),
        )
        return logit, h_new_list, z_new_list, aux
    # ── Full sequence forward ────────────────────────────────────────
    def forward(self, tokens: torch.Tensor):
        """
        tokens : (B, T)
        Returns (logits (B,T,V),  aux_losses dict)
        """
        B, T  = tokens.shape
        dev   = tokens.device
        h, z  = self.init_state(B, dev)
        logits_list  = []
        z_seq        = []                                   # for branch-crossing loss
        for t in range(T):
            lg, h, z, aux = self.step(tokens[:, t], h, z)
            logits_list.append(lg)
            z_seq.append(aux['z_final'][:, 0, :])          # head-0 only
        logits = torch.stack(logits_list, dim=1)            # (B, T, V)
        z_seq  = torch.stack(z_seq,  dim=1)                 # (B, T, 3) complex
        # ── Auxiliary losses ────────────────────────────────────────
        aux_losses = {}
        # Branch-crossing penalty
        aux_losses['branch_crossing'] = self.branch_monitor(z_seq)
        # φ-vacuum attraction loss (Fubini–Study distance to p_φ)
        pv       = phi_vacuum(dev).unsqueeze(0).expand(B, 3)
        vac_dist = fubini_study_distance(z_seq[:, -1, :], pv)
        aux_losses['vacuum_attraction'] = vac_dist.mean() * 0.001
        # GUE spectral regularisation over learnable 2-D weight matrices
        gue = tokens.new_zeros((), dtype=torch.float32)
        cnt = 0
        for name, p in self.named_parameters():
            if p.dim() == 2 and min(p.shape) >= 8:
                gue = gue + self.gue_reg(p)
                cnt += 1
        aux_losses['gue_spectral'] = gue / max(cnt, 1) * 0.001
        # φ-branch dominance: penalise antimatter-dominant tokens
        aux_losses['branch_stability'] = (
            F.relu(-(branch_magnitude(z_seq) - 1.0)).mean() * 0.01
        )
        return logits, aux_losses

# § TRAINER
# ─────────────────────────────────────────────────────────────────────────────
class CP2Trainer:
    """Standard LM trainer with CP² auxiliary losses."""
    def __init__(self, model: CP2ProjectiveSSM, lr: float = 3e-4):
        self.model = model
        self.opt   = torch.optim.AdamW(model.parameters(), lr=lr, weight_decay=0.01)
    def step(self, tokens: torch.Tensor) -> dict:
        """tokens: (B, T)  —  returns dict of scalar losses."""
        inp, tgt = tokens[:, :-1], tokens[:, 1:]
        logits, aux = self.model(inp)
        B, T, V = logits.shape
        lm_loss = F.cross_entropy(logits.reshape(B * T, V), tgt.reshape(B * T))
        total   = lm_loss + sum(aux.values())
        self.opt.zero_grad()
        total.backward()
        torch.nn.utils.clip_grad_norm_(self.model.parameters(), 1.0)
        self.opt.step()
        return {'lm': lm_loss.item(), 'total': total.item(), **{k: v.item() for k, v in aux.items()}}

# § DIAGNOSTICS
# ─────────────────────────────────────────────────────────────────────────────
def diagnose(model: CP2ProjectiveSSM, tokens: torch.Tensor) -> dict:
    """
    Run a sequence through the model and return CP² geometric diagnostics:
      - per-token branch magnitude (should hover near φ ≈ 1.618)
      - per-token FS distance from φ-vacuum
      - per-token generation weights across the 3 toric fixed points
      - per-token branch label (+1 matter / -1 antimatter)
    """
    model.eval()
    B, T  = tokens.shape
    dev   = tokens.device
    h, z  = model.init_state(B, dev)
    records = dict(branch_mag=[], vac_dist=[], gen_weights=[], branch_label=[])
    pv = phi_vacuum(dev)
    with torch.no_grad():
        for t in range(T):
            _, h, z, aux = model.step(tokens[:, t], h, z)
            z0 = aux['z_final'][:, 0, :]
            records['branch_mag'].append(aux['branch_mag'].cpu().numpy())
            records['vac_dist'].append(
                fubini_study_distance(z0, pv.unsqueeze(0).expand_as(z0)).cpu().numpy()
            )
            records['gen_weights'].append(aux['gen_weights'].cpu().numpy())
            records['branch_label'].append(aux['branch_label'].cpu().numpy())
    for k in records:
        records[k] = np.stack(records[k], axis=1)           # (B, T, ...)
    records['phi']               = PHI
    records['inv_phi']           = INV_PHI
    records['v_crit']            = V_CRIT
    records['inter_branch_dist'] = ARCTAN_HALF
    records['yang_mills_action'] = S_YM
    records['higgs_vev']         = HIGGS_VEV
    return records
